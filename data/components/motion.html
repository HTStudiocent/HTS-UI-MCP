进度与动效

三种用法：按钮随机推进、按住拖动实时改变（拖动时小球跟随显示位置）、输入百分比定位。动效参考：悬停 0.25s · 按压 0.15s · 渐显 0.6s。

随机推进（0.6s 平滑过渡）

拖动进度条 · 按住左右拖动实时改变（支持触屏），悬停/拖动时显示小球手柄

自定义进度 · 输入 0~100 的数字，进度条即时移动到对应位置，超出范围自动取 0 或 100

HTML
<div class="progress">
    <div class="progress-bar" style="width:60%"></div>
</div>

<!-- 可拖动 -->
<div class="progress progress-drag" id="dragBar">
    <div class="progress-bar" style="width:40%">
        <div class="progress-thumb"></div>
    </div>
</div>

<!-- 输入百分比 -->
<input class="input" id="pctInput" placeholder="输入百分比，如 60" inputmode="numeric">
<div class="progress">
    <div class="progress-bar" id="pctFill" style="width:0%"></div>
</div>

<hr class="divider">

JS
// 1. 随机推进（宽度变化由 CSS transition 0.6s 平滑过渡）
var bar = document.querySelector('#randomBar .progress-bar');
document.getElementById('runBtn').addEventListener('click', function () {
    bar.style.width = (20 + Math.floor(Math.random() * 80)) + '%';
});

// 2. 按住拖动实时改变（Pointer Events 兼容鼠标/触屏，拖动时小球放大）
var wrap = document.getElementById('dragBar');
var fill = wrap.querySelector('.progress-bar');
var dragging = false;
function setFromX(clientX) {
    var rect = wrap.getBoundingClientRect();
    var pct = Math.round((clientX - rect.left) / rect.width * 100);
    pct = Math.max(0, Math.min(100, pct));
    fill.style.width = pct + '%';
}
wrap.addEventListener('pointerdown', function (e) {
    dragging = true;
    wrap.classList.add('dragging'); // 显示并放大小球
    wrap.setPointerCapture(e.pointerId);
    setFromX(e.clientX);
});
wrap.addEventListener('pointermove', function (e) {
    if (dragging) setFromX(e.clientX);
});
wrap.addEventListener('pointerup', function () {
    dragging = false;
    wrap.classList.remove('dragging');
});

// 3. 输入百分比即时定位
var input = document.getElementById('pctInput');
var target = document.getElementById('pctFill');
input.addEventListener('input', function () {
    var v = parseInt(input.value, 10);
    if (isNaN(v)) return;
    v = Math.max(0, Math.min(100, v));
    target.style.width = v + '%';
});
