代码框

用于展示代码片段，带语言标签、行号与复制按钮；默认收起，点击标题栏展开；超长内容在框内滚动（最大高度由 --code-max-height 控制）。

HTML
<div class="code-box">
    <div class="code-box-head">
        <span class="code-box-lang">HTML</span>
        <button class="code-copy">复制</button>
    </div>
    <pre><code>// 你的代码</code></pre>
</div>

复制按钮交互（读取相邻 code 文本）：

JS
// 按行包裹（配合 CSS 计数器显示行号）
document.querySelectorAll('.code-box pre > code').forEach(function (code) {
    var lines = code.textContent.split('\n');
    code.textContent = '';
    lines.forEach(function (line) {
        var s = document.createElement('span');
        s.className = 'code-line';
        s.appendChild(document.createTextNode(line));
        code.appendChild(s);
    });
});

// 代码框展开/收起
document.querySelectorAll('.code-box').forEach(function (box) {
    var head = box.querySelector('.code-box-head');
    if (!head) return;
    head.addEventListener('click', function (e) {
        // 点击复制按钮不触发展开/收起
        if (e.target.classList.contains('code-copy')) return;
        box.classList.toggle('expanded');
    });
});

// 复制按钮（逐行取值拼回纯代码，不含行号；成功后短暂显示"已复制"）
document.querySelectorAll('.code-copy').forEach(function (btn) {
    btn.addEventListener('click', function () {
        var rows = btn.closest('.code-box').querySelectorAll('.code-line');
        var code = Array.prototype.map.call(rows, function (l) {
            return l.textContent;
        }).join('\n');
        function done() {
            var old = btn.textContent;
            btn.textContent = '已复制';
            setTimeout(function () { btn.textContent = old; }, 1200);
        }
        function fallback() {
            var t = document.createElement('textarea');
            t.value = code;
            document.body.appendChild(t);
            t.select();
            document.execCommand('copy');
            document.body.removeChild(t);
            done();
        }
        if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(code).then(done, fallback);
        } else {
            fallback();
        }
    });
});
