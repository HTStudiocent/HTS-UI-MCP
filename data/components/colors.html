色彩

只用黑白灰，禁止渐变与彩色。所有颜色通过 CSS 变量定义，切换主题自动适配：色块颜色、显示的色值、复制的值都随当前主题变化。点击色块复制。

    --block: #f5f5f5;       /* 区块背景 */
    --border: #e5e5e5;      /* 边框 */

    --text-soft: #555555;   /* 次要文字 */

    --text-faint: #999999;  /* 最浅文字 */
    --primary: #1a1a1a;     /* 主色 */
    --on-primary: #ffffff;  /* 主色上的文字 */
    --footer-bg: #1a1a1a;
    --footer-text: #ffffff;
    --footer-muted: #999999;
    --shadow: rgba(0,0,0,0.06);

}

JS
// 点击色块复制色值并弹出提示
document.querySelectorAll('.ui-swatch').forEach(function (sw) {
    sw.addEventListener('click', function () {
        var hex = sw.getAttribute('data-hex');
        navigator.clipboard.writeText(hex).then(function () {
            showToast('已复制 ' + hex);
        });
    });
});

// 主题切换时刷新色块：data-hex-light / data-hex-dark
function refreshSwatches() {
    var t = document.documentElement.getAttribute('data-theme');
    var dark = t ? t === 'dark'
        : window.matchMedia('(prefers-color-scheme: dark)').matches;
    document.querySelectorAll('.ui-swatch').forEach(function (sw) {
        var hex = sw.getAttribute(dark ? 'data-hex-dark' : 'data-hex-light');
        sw.setAttribute('data-hex', hex);
        var p = sw.querySelector('.hex');
        if (p) p.textContent = hex.toUpperCase();
    });
}
refreshSwatches();
document.addEventListener('hts-theme-change', refreshSwatches);
