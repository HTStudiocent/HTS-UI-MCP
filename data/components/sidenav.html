JS
// 侧边导航：桌面端缩回，窄屏抽屉式展开
var side = document.getElementById('uiSide');
var sideWrap = document.querySelector('.ui-side-wrap');
var sideToggle = document.getElementById('uiSideToggle');
var sideMask = document.getElementById('uiSideMask');

function isNarrow() { return window.innerWidth <= 800; }

function syncToggleIcon() {
    sideToggle.textContent = isNarrow()
        ? (sideWrap.classList.contains('open') ? '✕' : '☰')
        : (side.classList.contains('collapsed') ? '▶' : '◀');
}

function setDrawer(open) {
    sideWrap.classList.toggle('open', open);
    sideMask.classList.toggle('show', open);
    document.body.classList.toggle('drawer-open', open);
    syncToggleIcon();
}

sideToggle.addEventListener('click', function () {
    if (isNarrow()) {
        setDrawer(!sideWrap.classList.contains('open'));
    } else {
        side.classList.toggle('collapsed');
        syncToggleIcon();
    }
});
sideMask.addEventListener('click', function () { setDrawer(false); });
window.addEventListener('resize', syncToggleIcon);
syncToggleIcon();

// 滚动自动高亮（IntersectionObserver）
var links = document.querySelectorAll('.ui-side a');
var blocks = document.querySelectorAll('.ui-block');

// 窄屏点击导航项后收起抽屉
links.forEach(function (l) {
    l.addEventListener('click', function () {
        if (isNarrow()) setDrawer(false);
    });
});

if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (entries) {
        entries.forEach(function (en) {
            if (en.isIntersecting) {
                links.forEach(function (l) {
                    l.classList.toggle('active',
                        l.getAttribute('href') === '#' + en.target.id);
                });
            }
        });
    }, { rootMargin: '-30% 0px -60% 0px' });
    blocks.forEach(function (b) { io.observe(b); });
}
